rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function esAdmin() {
      return request.auth != null
        && request.auth.token.email == 'rgaraventa@gmail.com';
    }

    function emailVerificado() {
      return request.auth != null
        && request.auth.token.email_verified == true;
    }

    function esUsuarioEmpresa(tenantId, empresaId) {
      return request.auth != null &&
        get(/databases/$(database)/documents/tenants/$(tenantId)/usuarios/$(request.auth.uid)).data.empresaId == empresaId;
    }

    function pedidoValido() {
      let d = request.resource.data;
      return d.clienteUid == request.auth.uid
        && d.direccion is string && d.direccion.size() > 3 && d.direccion.size() < 300
        && d.items is list && d.items.size() > 0 && d.items.size() <= 20
        && (!('nota' in d)        || (d.nota is string        && d.nota.size() < 400))
        && (!('clienteNombre' in d) || (d.clienteNombre is string && d.clienteNombre.size() < 100))
        && (!('clienteTelefono' in d) || (d.clienteTelefono is string && d.clienteTelefono.size() < 30))
        && d.total is number && d.total > 0 && d.total < 10000000
        && d.estado == 'enviado';
    }

    match /tenants/{tenantId} {
      allow read: if true;
      allow write: if esAdmin();

      match /categorias/{catId} {
        allow read: if true;
        allow write: if esAdmin();
      }

      match /empresas/{empresaId} {
        allow read: if true;
        allow write: if esAdmin() || esUsuarioEmpresa(tenantId, empresaId);

        match /productos/{prodId} {
          allow read: if true;
          allow write: if esAdmin() || esUsuarioEmpresa(tenantId, empresaId);
        }
      }

      match /pedidos/{pedidoId} {
        // Solo cuentas de Google verificadas pueden crear pedidos
        // con validación de campos (anti-spam y anti-XSS en Firestore)
        allow create: if emailVerificado() && pedidoValido();

        allow read: if request.auth != null && (
          resource.data.clienteUid == request.auth.uid
          || esAdmin()
          || esUsuarioEmpresa(tenantId, resource.data.empresaId));

        allow update: if esAdmin() || esUsuarioEmpresa(tenantId, resource.data.empresaId);
        allow delete: if esAdmin();
      }

      match /usuarios/{uid} {
        allow read: if request.auth != null && (request.auth.uid == uid || esAdmin());
        allow create: if request.auth != null && request.auth.uid == uid
          && request.resource.data.rol == 'cliente'
          && !('empresaId' in request.resource.data);
        allow update: if esAdmin() || (request.auth.uid == uid
          && request.resource.data.rol == resource.data.rol
          && request.resource.data.get('empresaId','') == resource.data.get('empresaId',''));
        allow delete: if esAdmin();
      }
    }
  }
}
